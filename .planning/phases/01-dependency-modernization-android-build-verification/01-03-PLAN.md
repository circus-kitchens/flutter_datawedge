Replace the entire file content with:

```yaml
name: Pull Request
on:
  workflow_dispatch:
  pull_request:
    branches:
      - main
permissions:
  checks: write
  contents: read
env:
  FLUTTER_CHANNEL: "stable"
jobs:
  lint:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Flutter Setup
        uses: subosito/flutter-action@v2
        with:
          cache: true
          channel: ${{ env.FLUTTER_CHANNEL }}
      - name: Dependencies
        run: flutter pub get
      - name: Dart Analyze
        uses: invertase/github-action-dart-analyzer@v1
        with:
          fatal-infos: true
          fatal-warnings: true
          annotate: true
      - name: Check Formatting
        run: dart format --set-exit-if-changed --output none $(find lib test -name "*.dart" -not \( -name "*.freezed.dart" -o -name "*.g.dart" \))

  android-build:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    needs: [lint]
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Flutter Setup
        uses: subosito/flutter-action@v2
        with:
          cache: true
          channel: ${{ env.FLUTTER_CHANNEL }}
      - name: Java Setup
        uses: actions/setup-java@v4
        with:
          distribution: "temurin"
          java-version: "17"
      - name: Dependencies
        run: flutter pub get
      - name: Build Android Example APK
        run: cd example && flutter build apk --release

  test:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    needs: [lint]
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Flutter Setup
        uses: subosito/flutter-action@v2
        with:
          cache: true
          channel: ${{ env.FLUTTER_CHANNEL }}
      - name: Dependencies
        run: flutter pub get
      - name: Run Tests
        run: flutter test
```

Key decisions applied:
- FLUTTER_VERSION env var removed (per D-05, using stable channel dynamically)
- lint job updated to actions/checkout@v4 (from v3)
- dart format exclusion pattern updated to exclude *.freezed.dart and *.g.dart generated files
- android-build job: needs [lint], uses setup-java@v4, temurin JDK 17, runs `cd example && flutter build apk --release` (per D-06)
- test job: needs [lint], runs `flutter test` (per CI-05)
- Both android-build and test jobs run in parallel after lint passes
